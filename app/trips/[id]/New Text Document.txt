	"use client";

import { useEffect, useState } from "react";
import { useSearchParams, useRouter } from "next/navigation";

type ItineraryDay = {
  day: number;
  title: string;
  summary: string;
  morning: string;
  afternoon: string;
  evening: string;
};

type Trip = {
  destination: string;
  startDate: string;
  endDate: string;
  budget: string;
  style: string;
  interests: string[];
  days: ItineraryDay[];
};

export default function TripPage() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const id = searchParams.get("id");

  const [trip, setTrip] = useState<Trip | null>(null);

  useEffect(() => {
    if (!id) return;

    const stored = localStorage.getItem(`trip-${id}`);
    if (stored) {
      setTrip(JSON.parse(stored));
    } else {
      router.push("/plan");
    }
  }, [id, router]);

  if (!trip) {
    return (
      <main className="min-h-screen bg-black text-white flex items-center justify-center">
        <p className="text-slate-400">Loading your trip…</p>
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-black text-white px-4 py-10 fade-in">
      <div className="max-w-5xl mx-auto">
        <header className="mb-10">
          <h1 className="text-4xl font-bold mb-2">
            {trip.destination}: {trip.days.length}-day itinerary
          </h1>
          <p className="text-slate-300 text-sm">
            {trip.startDate} → {trip.endDate} • {trip.budget} budget • {trip.style} style
          </p>
          {trip.interests.length > 0 && (
            <p className="text-slate-400 text-sm mt-1">
              Interests: {trip.interests.join(", ")}
            </p>
          )}
        </header>

        <section className="space-y-6">
          {trip.days.map((day) => (
            <article
              key={day.day}
              className="rounded-xl border border-white/10 bg-white/5 p-6 fade-up"
            >
              <h2 className="text-2xl font-semibold mb-2">{day.title}</h2>
              <p className="text-slate-300 text-sm mb-4">{day.summary}</p>

              <div className="grid grid-cols-1 md:grid-cols-3 gap-6 text-sm">
                <div>
                  <p className="text-slate-400 uppercase text-xs mb-1">Morning</p>
                  <p>{day.morning}</p>
                </div>
                <div>
                  <p className="text-slate-400 uppercase text-xs mb-1">Afternoon</p>
                  <p>{day.afternoon}</p>
                </div>
                <div>
                  <p className="text-slate-400 uppercase text-xs mb-1">Evening</p>
                  <p>{day.evening}</p>
                </div>
              </div>
            </article>
          ))}
        </section>
      </div>
    </main>
  );
}
